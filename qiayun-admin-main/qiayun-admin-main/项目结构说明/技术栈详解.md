# qiayun-admin 技术栈详解

## 🏗️ 技术架构概览

qiayun-admin 采用现代化的前端技术栈，以 Vue 3 生态为核心，构建了一个完整的企业级管理后台解决方案。

## 🎯 核心技术栈

### 1. 前端框架 - Vue 3.4

**选择理由**：

- 性能提升：更快的渲染和更小的包体积
- Composition API：更好的逻辑复用和TypeScript支持
- `<script setup>`：更简洁的语法糖

### 7. HTTP 客户端 - Axios 1.7.2

**二次封装特性**：

- 请求/响应拦截器
- 错误统一处理
- Loading控制
- 请求取消

**封装示例**：

```typescript
// api/index.ts
class RequestHttp {
  service: AxiosInstance;

  constructor(config: AxiosRequestConfig) {
    this.service = axios.create(config);

    // 请求拦截器
    this.service.interceptors.request.use((config: CustomAxiosRequestConfig) => {
      // Token注入
      if (config.headers && typeof config.headers.set === "function") {
        config.headers.set("x-access-token", userStore.token);
      }

      // Loading控制
      config.loading && showFullScreenLoading();

      return config;
    });

    // 响应拦截器
    this.service.interceptors.response.use((response: AxiosResponse) => {
      const { data } = response;

      // 登录失效处理
      if (data.code == ResultEnum.OVERDUE) {
        userStore.setToken("");
        router.replace(LOGIN_URL);
        return Promise.reject(data);
      }

      return data;
    });
  }

  // 请求方法封装
  get<T>(url: string, params?: object): Promise<ResultData<T>> {
    return this.service.get(url, { params });
  }

  post<T>(url: string, params?: object): Promise<ResultData<T>> {
    return this.service.post(url, params);
  }
}
```

### 2. 主题切换机制

```typescript
// hooks/useTheme.ts
export const useTheme = () => {
  const globalStore = useGlobalStore();

  // 切换暗黑模式
  const switchDark = () => {
    const html = document.documentElement as HTMLElement;
    if (globalStore.isDark) {
      html.setAttribute("class", "dark");
    } else {
      html.setAttribute("class", "");
    }
  };

  // 修改主题色
  const changePrimary = (val: string) => {
    if (!val) return;

    const el = document.documentElement as HTMLElement;
    el.style.setProperty("--el-color-primary", val);

    // 生成渐变色
    for (let i = 1; i <= 9; i++) {
      const lightColor = getLightColor(val, i / 10);
      el.style.setProperty(`--el-color-primary-light-${i}`, lightColor);
    }

    globalStore.setGlobalState("primary", val);
  };

  return { switchDark, changePrimary };
};
```

### 1. 工具函数

```typescript
// utils/index.ts

// 深拷贝
export function deepClone<T>(obj: T): T {
  return JSON.parse(JSON.stringify(obj));
}

// 防抖
export function debounce<T extends (...args: any[]) => any>(func: T, delay: number): (...args: Parameters<T>) => void {
  let timeoutId: NodeJS.Timeout;
  return (...args: Parameters<T>) => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => func.apply(this, args), delay);
  };
}

// 节流
export function throttle<T extends (...args: any[]) => any>(func: T, delay: number): (...args: Parameters<T>) => void {
  let lastCall = 0;
  return (...args: Parameters<T>) => {
    const now = Date.now();
    if (now - lastCall >= delay) {
      lastCall = now;
      func.apply(this, args);
    }
  };
}
```

### 2. 自定义 Hooks

```typescript
// hooks/useTable.ts
export const useTable = (api: (params: any) => Promise<any>, initParam: object = {}, isPageable: boolean = true) => {
  const state = reactive({
    tableData: [],
    loading: false,
    pageable: {
      pageNum: 1,
      pageSize: 10,
      total: 0
    },
    searchParam: {},
    searchInitParam: initParam,
    totalParam: {}
  });

  // 获取表格数据
  const getTableList = async () => {
    try {
      state.loading = true;
      const { data } = await api(state.totalParam);

      if (isPageable) {
        state.tableData = data.list;
        state.pageable.total = data.total;
      } else {
        state.tableData = data;
      }
    } finally {
      state.loading = false;
    }
  };

  // 搜索
  const search = () => {
    state.pageable.pageNum = 1;
    updatedTotalParam();
    getTableList();
  };

  // 重置
  const reset = () => {
    state.pageable.pageNum = 1;
    state.searchParam = { ...state.searchInitParam };
    updatedTotalParam();
    getTableList();
  };

  return {
    ...toRefs(state),
    getTableList,
    search,
    reset
  };
};
```

  @if $breakpoint == mobile {
    @media (max-width: $mobile - 1) {

      @content;
    }
  }
